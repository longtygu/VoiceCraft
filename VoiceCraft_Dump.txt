using NAudio.Wave;
using System;

namespace VoiceCraft.App_Core.Services
{
    public class AudioPlayer : IDisposable
    {
        private WaveOut _outputDevice; // Đã đổi WaveOutEvent thành WaveOut
        private AudioFileReader _audioFile;

        public event EventHandler PlaybackStopped;

        public bool IsPlaying => _outputDevice != null && _outputDevice.PlaybackState == PlaybackState.Playing;
        public TimeSpan TotalTime => _audioFile?.TotalTime ?? TimeSpan.Zero;

        public TimeSpan CurrentTime
        {
            get => _audioFile?.CurrentTime ?? TimeSpan.Zero;
            set { if (_audioFile != null) _audioFile.CurrentTime = value; }
        }

        public void Load(string filePath)
        {
            DisposeOutDevice();
            try
            {
                _outputDevice = new WaveOut(); // Đã đổi thành WaveOut
                _outputDevice.PlaybackStopped += (s, e) => PlaybackStopped?.Invoke(this, EventArgs.Empty);
                _audioFile = new AudioFileReader(filePath);
                _outputDevice.Init(_audioFile);
            }
            catch (Exception ex)
            {
                System.Diagnostics.Debug.WriteLine($"Lỗi load âm thanh: {ex.Message}");
            }
        }

        public void Play() => _outputDevice?.Play();

        public void Pause() => _outputDevice?.Pause();

        public void Stop()
        {
            _outputDevice?.Stop();
            if (_audioFile != null) _audioFile.CurrentTime = TimeSpan.Zero;
        }

        private void DisposeOutDevice()
        {
            _outputDevice?.Stop();
            _outputDevice?.Dispose();
            _outputDevice = null;
            _audioFile?.Dispose();
            _audioFile = null;
        }

        public void Dispose() => DisposeOutDevice();
    }
}

using System;
using System.Diagnostics;
using System.IO;
using System.Net.Http;
using System.Threading.Tasks;

namespace VoiceCraft.App_Core.Services
{
    public class EngineManager : IDisposable
    {
        private Process _engineProcess;
        private readonly HttpClient _httpClient;
        private readonly string _enginePath;
        private readonly string _baseUrl = "http://127.0.0.1:8000";

        public EngineManager()
        {
            _httpClient = new HttpClient();
            _httpClient.Timeout = TimeSpan.FromSeconds(3); // Tránh kẹt timeout quá lâu
            _enginePath = FindEnginePath();
        }

        private string FindEnginePath()
        {
            string currentDir = AppDomain.CurrentDomain.BaseDirectory;
            for (int i = 0; i < 5; i++)
            {
                string checkPath = Path.Combine(currentDir, @"Engine\dist\TTS_Engine\TTS_Engine.exe");
                if (File.Exists(checkPath)) return Path.GetFullPath(checkPath);

                var parent = Directory.GetParent(currentDir);
                if (parent == null) break;
                currentDir = parent.FullName;
            }
            return string.Empty;
        }

        // Bổ sung hàm dọn dẹp các tiến trình ngầm đang chiếm cổng 8000
        private void KillZombieProcesses()
        {
            try
            {
                foreach (var process in Process.GetProcessesByName("TTS_Engine"))
                {
                    process.Kill();
                    process.WaitForExit(1000);
                }
            }
            catch (Exception ex) { Logger.Error("Lỗi dọn dẹp Zombie Process", ex); }
        }

        public async Task<bool> StartEngineAsync()
        {
            KillZombieProcesses(); // Dọn dẹp trước khi chạy

            if (string.IsNullOrEmpty(_enginePath) || !File.Exists(_enginePath))
            {
                throw new FileNotFoundException("LỖI: Chưa build Python Engine!");
            }

            _engineProcess = Process.Start(new ProcessStartInfo
            {
                FileName = _enginePath,
                UseShellExecute = false,
                CreateNoWindow = true,
                WindowStyle = ProcessWindowStyle.Hidden
            });

            return await WaitForEngineReadyAsync();
        }

        private async Task<bool> WaitForEngineReadyAsync()
        {
            for (int i = 0; i < 15; i++)
            {
                try
                {
                    var response = await _httpClient.GetAsync($"{_baseUrl}/ping");
                    if (response.IsSuccessStatusCode) return true;
                }
                catch (Exception) { } // Bỏ qua lỗi kết nối để thử lại
                await Task.Delay(1000);
            }
            return false;
        }

        public void StopEngine()
        {
            if (_engineProcess != null && !_engineProcess.HasExited)
            {
                try
                {
                    _engineProcess.Kill();
                    _engineProcess.WaitForExit(2000);
                }
                catch (Exception) { }
            }
        }

        public void Dispose()
        {
            StopEngine();
            _httpClient?.Dispose();
            _engineProcess?.Dispose();
        }
    }
}

using System;
using System.IO;

namespace VoiceCraft.App_Core.Services
{
    public static class Logger
    {
        private static readonly string LogDir;
        private static readonly string LogFile;

        static Logger()
        {
            string baseDir = AppDomain.CurrentDomain.BaseDirectory;
            LogDir = Path.GetFullPath(Path.Combine(baseDir, @"..\Logs"));
            if (!Directory.Exists(LogDir)) Directory.CreateDirectory(LogDir);
            LogFile = Path.Combine(LogDir, $"VoiceCraft_{DateTime.Now:yyyyMMdd}.log");
        }

        public static void Info(string message) => Write("INFO", message);

        public static void Error(string message, Exception ex = null)
            => Write("ERROR", $"{message}{(ex != null ? $" | Chi tiết: {ex.Message}" : "")}");

        private static void Write(string level, string message)
        {
            try
            {
                string logEntry = $"[{DateTime.Now:HH:mm:ss}] [{level}] {message}\n";
                File.AppendAllText(LogFile, logEntry);
            }
            catch (Exception) { /* Bỏ qua nếu không ghi được để tránh crash app */ }
        }
    }
}

        private async Task SynthesizeAsync()
        {
            IsProcessing = true;
            Logger.Info($"Gửi yêu cầu TTS. Giọng: {SelectedVoice}, Tốc độ: {Speed}, Độ dài text: {CharCount}");

        private PointCollection GenerateWaveform(string filePath)
        {
            var points = new PointCollection();
            if (string.IsNullOrEmpty(filePath) || !File.Exists(filePath)) return points;

            double width = 1000, height = 100, midY = height / 2;

            try
            {
                // Dùng WaveFileReader thay vì AudioFileReader để ép đọc byte nguyên thủy 100% an toàn
                using (var reader = new WaveFileReader(filePath))
                {
                    int bytesPerSample = reader.WaveFormat.BitsPerSample / 8; // Mặc định 2 bytes (16-bit PCM)
                    long totalSamples = reader.Length / bytesPerSample;
                    int samplesPerPixel = Math.Max(1, (int)(totalSamples / width));
                    int readChunkSize = samplesPerPixel * reader.WaveFormat.Channels * bytesPerSample;

                    byte[] buffer = new byte[readChunkSize];

                    var topPoints = new List<Point>();
                    var bottomPoints = new List<Point>();

                    int x = 0, bytesRead;
                    // Lệnh Read(byte[], int, int) được gọi trực tiếp từ class gốc Stream
                    while ((bytesRead = reader.Read(buffer, 0, buffer.Length)) > 0 && x <= width)
                    {
                        float max = 0;
                        // Phân tích dữ liệu 16-bit ra thành mức độ sóng âm
                        for (int i = 0; i < bytesRead - 1; i += 2)
                        {
                            short sample = BitConverter.ToInt16(buffer, i);
                            float val = Math.Abs((float)sample / 32768f);
                            if (val > max) max = val;
                        }
                        topPoints.Add(new Point(x, midY - (max * midY)));
                        bottomPoints.Insert(0, new Point(x, midY + (max * midY)));
                        x++;
                    }

                    points.Add(new Point(0, midY));
                    foreach (var p in topPoints) points.Add(p);
                    foreach (var p in bottomPoints) points.Add(p);
                    points.Add(new Point(0, midY));
                }
            }
            catch (Exception ex) { Logger.Error("Lỗi vẽ waveform UI", ex); }

            return points;
        }

import sys
import os

if sys.stdout is None:
    sys.stdout = open(os.devnull, "w")
if sys.stderr is None:
    sys.stderr = open(os.devnull, "w")
if sys.stdin is None:
    sys.stdin = open(os.devnull, "r")

import asyncio
import wave
import uvicorn
from fastapi import FastAPI
from pydantic import BaseModel
from fastapi.responses import FileResponse
from piper import PiperVoice

app = FastAPI(title="VoiceCraft Engine")

class TTSRequest(BaseModel):
    text: str
    voice_id: str
    speed: float = 1.0
    pitch: float = 1.0

base_dir = os.path.dirname(os.path.abspath(__file__))
models_dir = os.path.abspath(os.path.join(base_dir, "..", "Models"))
output_dir = os.path.abspath(os.path.join(base_dir, "..", "Output"))
os.makedirs(models_dir, exist_ok=True)
os.makedirs(output_dir, exist_ok=True)

loaded_voices = {}

def get_voice(voice_id: str) -> PiperVoice:
    if voice_id in loaded_voices:
        return loaded_voices[voice_id]
        
    model_name = "vi-VN-model.onnx" 
    model_path = os.path.join(models_dir, model_name)
    
    if not os.path.exists(model_path):
        raise FileNotFoundError(f"Chưa có model tại {model_path}")
        
    voice = PiperVoice.load(model_path)
    loaded_voices[voice_id] = voice
    return voice

@app.get("/ping")
async def ping():
    return {"status": "ready"}

@app.post("/synthesize")
async def synthesize(req: TTSRequest):
    try:
        voice = await asyncio.to_thread(get_voice, req.voice_id)
        file_path = os.path.join(output_dir, "temp_output.wav")
        
        with wave.open(file_path, "wb") as wav_file:
            length_scale = 1.0 / req.speed 
            wav_file.setnchannels(1)
            wav_file.setsampwidth(2)
            wav_file.setframerate(voice.config.sample_rate)
            
            await asyncio.to_thread(
                voice.synthesize, req.text, wav_file, length_scale=length_scale
            )
            
        return FileResponse(file_path, media_type="audio/wav", filename="output.wav")
        
    except Exception as e:
        return {"status": "error", "message": str(e)}

async def main():
    config = uvicorn.Config(
        app=app, 
        host="127.0.0.1", 
        port=8080,
        log_config=None,
        access_log=False,
        use_colors=False
    )
    server = uvicorn.Server(config)
    await server.serve()

if __name__ == "__main__":
    asyncio.run(main())
